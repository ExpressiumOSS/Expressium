using Microsoft.Playwright;
using System;
using System.Threading;
using System.Threading.Tasks;

namespace $Namespace$.Tests.Infrastructure
{
    public enum BrowserTypes
    {
        Chrome,
        Chromium,
        Firefox,
        Edge,
        Webkit
    }

    public class BrowserFactory
    {
        private static IBrowser browser;
        private static readonly SemaphoreSlim initLock = new SemaphoreSlim(1, 1);

        public static async Task<(IPage page, IBrowserContext context)> Initialize(Configuration configuration)
        {
            var playwright = await PlaywrightProvider.GetInstanceAsync();

            await initLock.WaitAsync();
            try
            {
                if (browser == null)
                    browser = await GetBrowser(playwright, configuration);
            }
            finally
            {
                initLock.Release();
            }

            var context = await GetBrowserContext(browser, configuration);

            if (configuration.Traces)
            {
                await context.Tracing.StartAsync(new TracingStartOptions
                {
                    Screenshots = true,
                    Snapshots = true,
                    Sources = true
                });
            }

            var page = await context.NewPageAsync();
            page.SetDefaultTimeout(configuration.PageTimeout);

            if (configuration.Url != null)
                await page.GotoAsync(configuration.Url);

            return (page, context);
        }

        private static async Task<IBrowser> GetBrowser(IPlaywright playwright, Configuration configuration)
        {
            if (!Enum.TryParse<BrowserTypes>(configuration.BrowserType, ignoreCase: true, out var browserType))
                throw new ArgumentException($"Specified browser type '{configuration.BrowserType}' is unknown...");

            switch (browserType)
            {
                case BrowserTypes.Chrome:
                case BrowserTypes.Chromium:
                case BrowserTypes.Edge:
                    return await playwright.Chromium.LaunchAsync(GetBrowserTypeLaunchOptions(configuration));
                case BrowserTypes.Firefox:
                    return await playwright.Firefox.LaunchAsync(GetBrowserTypeLaunchOptions(configuration));
                case BrowserTypes.Webkit:
                    return await playwright.Webkit.LaunchAsync(GetBrowserTypeLaunchOptions(configuration));
                default:
                    throw new ArgumentException($"Specified browser type '{configuration.BrowserType}' is unknown...");
            }
        }

        private static async Task<IBrowserContext> GetBrowserContext(IBrowser browser, Configuration configuration)
        {
            var options = new BrowserNewContextOptions
            {
                ViewportSize = configuration.Headless
                    ? new ViewportSize { Width = configuration.WindowWidth, Height = configuration.WindowHeight }
                    : ViewportSize.NoViewport
            };

            if (configuration.Videos)
            {
                options.RecordVideoDir = configuration.VideosPath;
                options.RecordVideoSize = new RecordVideoSize
                {
                    Width = configuration.WindowWidth,
                    Height = configuration.WindowHeight
                };
            }

            return await browser.NewContextAsync(options);
        }

        private static BrowserTypeLaunchOptions GetBrowserTypeLaunchOptions(Configuration configuration)
        {
            var options = new BrowserTypeLaunchOptions
            {
                Headless = configuration.Headless,
                Timeout = configuration.BrowserTimeout,
                Channel = GetLaunchOptionsChannel(configuration)
            };

            if (configuration.Maximize)
                options.Args = new[] { "--start-maximized" };

            return options;
        }

        private static string GetLaunchOptionsChannel(Configuration configuration)
        {
            if (!Enum.TryParse<BrowserTypes>(configuration.BrowserType, ignoreCase: true, out var browserType))
                return null;

            return browserType switch
            {
                BrowserTypes.Chrome => "chrome",
                BrowserTypes.Edge => "msedge",
                BrowserTypes.Chromium => null,
                BrowserTypes.Firefox => null,
                BrowserTypes.Webkit => null,
                _ => null
            };
        }

        public static async Task Dispose()
        {
            await initLock.WaitAsync();
            try
            {
                if (browser != null)
                {
                    await browser.CloseAsync();
                    browser = null;
                }
            }
            finally
            {
                initLock.Release();
            }
        }
    }
}
