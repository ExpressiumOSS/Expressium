using Microsoft.Playwright;
using System.Threading.Tasks;

namespace $Namespace$.Controls
{
    public class WebControl
    {
        protected IPage page;
        protected ILocator locator;
        protected ILocator childLocator;

        public WebControl(IPage driver, ILocator locator)
        {
            this.page = driver;
            this.locator = locator;
        }

        public async Task WaitForPageElementIsVisible()
        {
            await locator.WaitForElementIsVisible();
        }

        public async Task WaitForPageElementIsEnabled()
        {
            await locator.WaitForElementIsEnabled();
        }

        public async Task<bool> IsVisible()
        {
            return await locator.IsVisibleAsync();
        }

        public async Task<bool> IsEnabled()
        {
            return await locator.IsEnabledAsync();
        }

        public async Task<bool> IsHidden()
        {
            return await locator.IsHiddenAsync();
        }
    }
}
