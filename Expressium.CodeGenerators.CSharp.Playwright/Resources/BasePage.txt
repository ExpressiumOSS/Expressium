using log4net;
using Microsoft.Playwright;
using System.Threading.Tasks;
using System;
using $Namespace$.Controls;

namespace $Namespace$.Pages
{
    public class BasePage
    {
        protected ILog logger;
        protected IPage page;

        public static int PageReadyStateTimeout { get; set; }

        public static string PageSpinnerSelector { get; set; }
        public static int PageSpinnerInitialTimeout { get; set; }
        public static int PageSpinnerTimeout { get; set; }

        static BasePage()
        {
            PageReadyStateTimeout = 10000;

            PageSpinnerSelector = null;
            PageSpinnerInitialTimeout = 150;
            PageSpinnerTimeout = 10000;
        }

        public BasePage(ILog logger, IPage page)
        {
            this.logger = logger;
            this.page = page;
        }

        public async Task<string> GetTitle()
        {
            logger.Info("GetTitle()");
            return await page.TitleAsync();
        }

        public string GetURL()
        {
            logger.Info("GetURL()");
            return new Uri(page.Url).LocalPath;
        }

        protected async Task WaitForPageDomContentLoaded()
        {
            await page.WaitForLoadStateAsync(LoadState.DOMContentLoaded, new() { Timeout = PageReadyStateTimeout });
        }

        protected async Task WaitForPageSpinnerIsCompleted()
        {
            if (PageSpinnerSelector != null)
            {
                await Task.Delay(PageSpinnerInitialTimeout);
                await page.Locator(PageSpinnerSelector).WaitForAsync(new() { State = WaitForSelectorState.Hidden, Timeout = PageSpinnerTimeout });
            }
        }

        protected async Task WaitForPageTitleEquals(string title)
        {
            await page.WaitForFunctionAsync($"() => document.title === '{title}'");
        }

        protected async Task WaitForPageTitleContains(string title)
        {
            await page.WaitForFunctionAsync($"() => document.title.includes('{title}')");
        }

        protected async Task WaitForPageUrlEquals(string urlPath)
        {
            await page.WaitForFunctionAsync($"() => window.location.pathname === '{urlPath}'");
        }

        protected async Task WaitForPageUrlContains(string urlPath)
        {
            await page.WaitForFunctionAsync($"() => window.location.pathname.includes('{urlPath}')");
        }

        protected async Task WaitForPageElementIsVisible(WebControl control)
        {
            await control.WaitForPageElementIsVisible();
        }

        protected async Task WaitForPageElementIsEnabled(WebControl control)
        {
            await control.WaitForPageElementIsEnabled();
        }

        public async Task NavigateBack()
        {
            logger.Info("NavigateBack()");
            await page.GoBackAsync();
        }

        public async Task NavigateForward()
        {
            logger.Info("NavigateForward()");
            await page.GoForwardAsync();
        }

        public async Task NavigateRefresh()
        {
            logger.Info("NavigateRefresh()");
            await page.ReloadAsync();
        }

        public async Task ScrollToTop()
        {
            logger.Info("ScrollToTop()");
            await page.EvaluateAsync("window.scrollTo(0, 0);");
        }

        public async Task ScrollToBottom()
        {
            logger.Info("ScrollToBottom()");
            await page.EvaluateAsync("window.scrollTo(0, document.body.scrollHeight);");
        }

        public void SwitchToAlertAccept()
        {
        }

        public void SwitchToAlertDismiss()
        {
        }

        public int GetNumberOfBrowserTabs()
        {
            int count = page.Context.Pages.Count;
            logger.Info($"GetNumberOfBrowserTabs() [{count}]");
            return count;
        }

        public void WaitForNewBrowserTabIsOpened(int currentTabCount)
        {
        }

        public void SelectBrowserTab(int index)
        {
        }
    }
}
