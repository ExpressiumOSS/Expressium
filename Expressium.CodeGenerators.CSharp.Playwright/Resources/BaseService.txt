using log4net;
using Microsoft.Playwright;
using System;
using System.Collections.Generic;
using System.Diagnostics;
using System.Text.Encodings.Web;
using System.Text.Json;
using System.Threading.Tasks;

namespace $Namespace$.Services
{
    public class BaseService
    {
        private const string CorrelationIdHeader = "X-Correlation-Id";
        private const string ContentTypeHeader = "Content-Type";
        private const string JsonContentType = "application/json";
        private const string RedactedValue = "***REDACTED***";

        private static readonly HashSet<string> RedactedHeaders = new HashSet<string>(StringComparer.OrdinalIgnoreCase)
        {
            "Authorization",
            "X-Api-Key",
            "Cookie",
            "Set-Cookie"
        };

        private static readonly JsonSerializerOptions PrettyPrintOptions = new JsonSerializerOptions
        {
            WriteIndented = true,
            Encoder = JavaScriptEncoder.UnsafeRelaxedJsonEscaping
        };

        protected readonly ILog logger;
        protected readonly IAPIRequestContext request;
        protected readonly Dictionary<string, string> headers;

        public bool IncludeRequestHeadersLogging { get; set; } = true;
        public bool IncludeRequestBodyLogging { get; set; } = true;
        public bool IncludeResponseHeadersLogging { get; set; } = true;
        public bool IncludeResponseBodyLogging { get; set; } = true;

        public BaseService(ILog logger, IAPIRequestContext request, Dictionary<string, string> headers = null)
        {
            this.logger = logger ?? throw new ArgumentNullException(nameof(logger));
            this.request = request ?? throw new ArgumentNullException(nameof(request));
            this.headers = new Dictionary<string, string>(headers ?? new Dictionary<string, string>(), StringComparer.OrdinalIgnoreCase);
        }

        protected Task<IAPIResponse> GetAsync(string endpoint, APIRequestContextOptions options = null) => SendAsync("GET", endpoint, options);
        protected Task<IAPIResponse> PostAsync(string endpoint, APIRequestContextOptions options = null) => SendAsync("POST", endpoint, options);
        protected Task<IAPIResponse> PutAsync(string endpoint, APIRequestContextOptions options = null) => SendAsync("PUT", endpoint, options);
        protected Task<IAPIResponse> DeleteAsync(string endpoint, APIRequestContextOptions options = null) => SendAsync("DELETE", endpoint, options);

        protected async Task<T> DeserializeAsync<T>(IAPIResponse response)
        {
            return await response.JsonAsync<T>();
        }

        private async Task<IAPIResponse> SendAsync(string method, string endpoint, APIRequestContextOptions options)
        {
            options ??= new APIRequestContextOptions();
            options.Method = method;
            options.Headers = BuildHeaders(options);

            LogRequest(endpoint, options);

            var stopwatch = Stopwatch.StartNew();
            var response = await request.FetchAsync(endpoint, options);
            stopwatch.Stop();

            await LogResponseAsync(response, stopwatch.Elapsed);

            return response;
        }

        // Priority: per-call headers, then service-level headers, then defaults
        private Dictionary<string, string> BuildHeaders(APIRequestContextOptions options)
        {
            var result = new Dictionary<string, string>(StringComparer.OrdinalIgnoreCase);

            if (options.Headers != null)
            {
                foreach (var header in options.Headers)
                    result[header.Key] = header.Value;
            }

            foreach (var header in headers)
                result.TryAdd(header.Key, header.Value);

            result.TryAdd(CorrelationIdHeader, Guid.NewGuid().ToString());

            if (options.DataObject != null || options.DataString != null)
                result.TryAdd(ContentTypeHeader, JsonContentType);

            return result;
        }

        private void LogRequest(string endpoint, APIRequestContextOptions options)
        {
            logger.Debug("");
            logger.Debug("[REQUEST]");
            logger.Debug($"Method: {options.Method}");
            logger.Debug($"URL: {endpoint}");

            if (IncludeRequestHeadersLogging)
                LogHeaders(options.Headers);

            if (IncludeRequestBodyLogging)
                LogBody(options.DataString ?? SerializeOrNull(options.DataObject));
        }

        private async Task LogResponseAsync(IAPIResponse response, TimeSpan duration)
        {
            logger.Debug("");
            logger.Debug("[RESPONSE]");
            logger.Debug($"Status: {response.Status} {response.StatusText}");
            logger.Debug($"Duration: {(long)duration.TotalMilliseconds} ms");

            if (IncludeResponseHeadersLogging)
                LogHeaders(response.Headers);

            if (IncludeResponseBodyLogging)
                LogBody(PrettyPrintOrRaw(await response.TextAsync()));
        }

        private void LogHeaders(IEnumerable<KeyValuePair<string, string>> headersToLog)
        {
            if (headersToLog == null)
                return;

            logger.Debug("Headers:");
            foreach (var header in headersToLog)
            {
                var value = RedactedHeaders.Contains(header.Key) ? RedactedValue : header.Value;
                logger.Debug($"  {header.Key} = {value}");
            }
        }

        private void LogBody(string body)
        {
            if (string.IsNullOrWhiteSpace(body))
                return;

            logger.Debug("Body:");
            foreach (var line in body.Replace("\r\n", "\n").Split('\n'))
                logger.Debug(line);
        }

        private static string SerializeOrNull(object value)
        {
            return value == null ? null : JsonSerializer.Serialize(value, PrettyPrintOptions);
        }

        private static string PrettyPrintOrRaw(string text)
        {
            if (string.IsNullOrWhiteSpace(text))
                return null;

            try
            {
                return JsonSerializer.Serialize(JsonSerializer.Deserialize<JsonElement>(text), PrettyPrintOptions);
            }
            catch (JsonException)
            {
                return text;
            }
        }
    }
}
